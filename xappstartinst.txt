# --- BEFORE EVERY LOCAL BUILD ---

# 1. Set config.ts BASE_URL to production
#    src/constants/config.ts → BASE_URL: "https://api.curelihealth.com"

# 2. Bump versionCode in app.config.js (must be higher than last Play Store upload)
#    android: { versionCode: X }   ← increment by 1 each build
#    Current production versionCode on Play Store: check Play Console →
#    App Bundle Explorer → highest version code number

# 3. Confirm google-services.json exists
ls google-services.json

# 4. Remove google-services.json from .gitignore temporarily
#    (EAS local build archives the project and gitignored files are excluded)
sed -i '' '/google-services.json/d' .gitignore


# --- BUILD ---

cd cureli-mobile
eas build --profile production --platform android --local

# Build takes ~15-20 minutes on first run, faster after (Gradle cache)
# Output .aab file path is printed at the end, something like:
# /Users/sibixavier/Projects/cureli-erp-devs/cureli-mobile/build-XXXXXXXXX.aab


# --- AFTER BUILD ---

# Add google-services.json back to .gitignore

# ============================================================
# WHEN TO REBUILD vs WHEN TO JUST RELOAD
# ============================================================

# Just reload Metro (press r) or restart Metro:
#   - changed .ts .tsx .js .jsx files
#   - changed hooks, screens, components, stores, services
#   - changed src/constants/config.ts

# Full rebuild required (npx expo run:android or run:ios):
#   - changed app.config.js
#   - changed eas.json
#   - added/removed a native package (npm install something native)
#   - changed android permissions
#   - changed iOS permissions or infoPlist
#   - changed google-services.json
#   - changed expo-notifications plugin config
#   - changed expo-location plugin config
#   - changed splash screen or icon
#   - changed scheme or package/bundle identifier
#   - ran npx expo prebuild --clean


# ============================================================
# TROUBLESHOOTING
# ============================================================

# API calls fail on Android USB:
#   Make sure you ran:
#   adb reverse tcp:5000 tcp:5000
#   and config.ts BASE_URL is http://localhost:5000

# API calls fail on iPhone:
#   Do NOT use localhost on iPhone
#   Use your machine LAN IP: http://192.168.x.x:5000
#   Run ipconfig to find your current IP
#   Make sure phone and computer are on same WiFi

# App won't open after Metro starts:
#   Dev build is not installed — run:
#   npx expo run:android
#   (or npx expo run:ios on macOS)

# Changes not reflecting:
#   npx expo start --dev-client -c

# "No development build installed" error:
#   npx expo run:android
#   npx expo start --dev-client

# After prebuild --clean, app disappears from phone:
#   Expected — run npx expo run:android to reinstall

# Local build fails — google-services.json not found:
#   The file is gitignored and gets excluded from the build archive
#   Fix: sed -i '' '/google-services.json/d' .gitignore
#   Then retry build, then add it back after:
#   echo "google-services.json" >> .gitignore

# Local build fails — ANDROID_HOME not set:
#   source ~/.zshrc
#   echo $ANDROID_HOME   ← should not be empty

# Local build fails — Java not found:
#   java --version
#   If missing: brew install --cask temurin@17
#   Then: echo 'export JAVA_HOME=$(/usr/libexec/java_home -v 17)' >> ~/.zshrc
#   source ~/.zshrc

# Local build fails — credentials.json missing:
#   Recreate it at cureli-mobile/credentials.json
#   Get keystore passwords from 1Password / team vault
#   Keystore file is at ~/cureli-keys/cureli.jks
#   If keystore file missing: eas credentials --platform android →
#   production → Download existing keystore → move to ~/cureli-keys/cureli.jks
